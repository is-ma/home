## PREFIX = Ctrl+Space (antes Ctrl+b)
## Para enviar PREFIX a la shell: PREFIX PREFIX
##
## Organizado por orden alfabético (segunda tecla del binding).

# [!]
PREFIX !    # Romper panel actual en ventana nueva (break-pane)

# ["]
PREFIX "    # Dividir panel horizontalmente (split-window -v)

# [$]
$           # Fin de línea (end-of-line) [vi]
PREFIX $    # Renombrar sesión actual (rename-session)

# [%]
%           # Ir al siguiente corchete/parentesis coincidente
PREFIX %    # Dividir panel verticalmente (split-window -h)

# [&]
PREFIX &    # Matar ventana actual (kill-window)

# [']
PREFIX '    # Pedir índice de ventana para seleccionar (command-prompt)

# [(]
PREFIX (    # Cambiar a sesión anterior (switch-client -p)

# [)]
PREFIX )    # Cambiar a sesión siguiente (switch-client -n)

# [,]
,           # Repetir último salto en dirección inversa (jump-reverse)
PREFIX ,    # Renombrar ventana actual (rename-window)

# [-]
PREFIX -    # Borrar el buffer copiado más reciente (delete-buffer)

# [.]
PREFIX .    # Pedir índice para mover ventana actual (move-window)

# [/]
PREFIX /    # Buscar hacia adelante en modo copia (search-forward)

# [:]
:           # Ir a línea específica (goto-line)
PREFIX :    # Abrir prompt de comando de tmux (command-prompt)

# [;]
;           # Repetir último salto (jump-again)
PREFIX ;    # Ir al panel previamente activo (last-pane)

# [0-9]
PREFIX 0    # Ventana 10 (configurado)
PREFIX 1..9 # Seleccionar ventana 1 a 9 (select-window)
0           # Inicio de línea (start-of-line) [vi]

# [=]
PREFIX =    # Elegir buffer interactivamente para pegar (choose-buffer)

# [>]
M->         # Ir al final del historial (history-bottom) [vi/emacs]

# [?]
PREFIX ?    # Listar todos los atajos de teclado (list-keys)

# [@]

# [A]
A           # Acumular selección en buffer 'accum' SIN salir [vi] (custom)
C-a         # Inicio de línea (start-of-line) [emacs]

# [B]
B           # Palabra anterior separada por espacios (previous-space) [vi]
b           # Palabra anterior (previous-word) [vi]
M-b         # Palabra anterior (previous-word) [emacs]
C-b         # Página arriba (page-up) [vi]
M-C-b       # Ir al corchete/parentesis coincidente anterior [emacs]

# [C]
PREFIX c    # Crear nueva ventana (new-window)
C-c         # Cancelar operación / salir (cancel) [emacs]

# [D]
PREFIX D    # Elegir cliente para desconectar (choose-client)
PREFIX d    # Desconectar / detach sesión (detach-client)
Down        # Cursor abajo (cursor-down)
C-d         # Media página abajo (halfpage-down) [vi]
M-Down      # Media página abajo (halfpage-down) [emacs]
C-Down      # Scroll hacia abajo (scroll-down) [emacs]

# [E]
E           # Fin de palabra separada por espacios (next-space-end) [vi]
e           # Fin de palabra actual/siguiente (next-word-end) [vi]
Enter       # Copiar selección y salir (copy-selection) [vi]
Escape      # Salir del modo copia (cancel)
C-e         # Fin de línea (end-of-line) [emacs] / Scrollear abajo

# [F]
F           # Saltar hacia atrás a carácter (jump-backward) [vi]
f           # Saltar hacia adelante a carácter (jump-forward) [vi]
PREFIX f    # Buscar texto en ventanas abiertas (find-window)
M-f         # Fin de palabra actual/siguiente (next-word-end) [emacs]

# [G]
g           # Ir a línea específica [vi] / Ir al inicio del historial
G           # Ir al final del historial (history-bottom) [vi]
C-g         # Cancelar operación / salir (cancel) [emacs]

# [H]
h           # Cursor izquierda (cursor-left)
H           # Ir a línea superior visible (top-line) [vi]
M-h         # Pane navigation: mover a panel izquierda (configurado)
M-H         # Resize: panel izquierda 1 celda (configurado)
PREFIX C-Left  # Redimensionar panel izquierda 1 celda (resize-pane -L)
PREFIX M-Left  # Redimensionar panel izquierda 5 celdas (resize-pane -L)

# [I]
PREFIX i    # Mostrar información de la ventana actual (display-message)

# [J]
j           # Cursor abajo (cursor-down)
M-j         # Pane navigation: mover a panel abajo (configurado)
M-J         # Resize: panel abajo 1 celda (configurado)
PREFIX C-Down  # Redimensionar panel abajo 1 celda (resize-pane -D)
PREFIX M-Down  # Redimensionar panel abajo 5 celdas (resize-pane -D)
PREFIX Down    # Cambiar foco al panel de abajo (select-pane -D)

# [K]
k           # Cursor arriba (cursor-up)
M-k         # Pane navigation: mover a panel arriba (configurado)
M-K         # Resize: panel arriba 1 celda (configurado)
PREFIX C-Up    # Redimensionar panel arriba 1 celda (resize-pane -U)
PREFIX M-Up    # Redimensionar panel arriba 5 celdas (resize-pane -U)
PREFIX Up      # Cambiar foco al panel de arriba (select-pane -U)

# [L]
L           # Ir a línea inferior visible (bottom-line) [vi]
l           # Cursor derecha (cursor-right)
Left        # Cursor izquierda (cursor-left)
PREFIX l    # Cambiar a la ventana previamente seleccionada (last-window)
PREFIX L    # Volver a la última sesión usada (switch-client -l)
PREFIX Left # Cambiar foco al panel de la izquierda (select-pane -L)
M-l         # Centrar cursor horizontalmente (cursor-centre-horizontal) [emacs]
C-l         # Centrar cursor verticalmente en pantalla [emacs]
M-l         # Pane navigation: mover a panel derecha (configurado)
M-L         # Resize: panel derecha 1 celda (configurado)
PREFIX C-Right # Redimensionar panel derecha 1 celda (resize-pane -R)
PREFIX M-Right # Redimensionar panel derecha 5 celdas (resize-pane -R)
PREFIX Right   # Cambiar foco al panel de la derecha (select-pane -R)

# [M]
PREFIX m    # Marcar panel actual (select-pane -m)
PREFIX M    # Limpiar marca del panel (select-pane -M)
M           # Ir a línea media visible (middle-line) [vi]
M-m         # Ir al primer carácter no blanco (back-to-indentation) [emacs]
M-r         # Ir a línea media visible (middle-line) [emacs]
M-R         # Ir a línea superior visible (top-line) [emacs]
M-<         # Ir al inicio del historial (history-top) [emacs]

# [N]
N           # Coincidencia anterior de búsqueda (search-reverse)
n           # Siguiente coincidencia de búsqueda (search-again)
PREFIX n    # Cambiar a ventana siguiente (next-window)
PREFIX M-n  # Ir a siguiente ventana con bell/actividad
C-n         # Cursor abajo (cursor-down) [emacs]

# [O]
o           # Cambiar extremo de la selección (other-end) [vi]
PREFIX o    # Seleccionar siguiente panel (select-pane -t :.+)
PREFIX M-o  # Rotar paneles hacia atrás (rotate-window -D)
PREFIX C-o  # Rotar paneles hacia adelante (rotate-window)

# [P]
P           # Mostrar/ocultar indicador de posición (toggle-position)
PREFIX p    # Cambiar a ventana anterior (previous-window)
PREFIX M-p  # Ir a anterior ventana con bell/actividad
C-p         # Cursor arriba (cursor-up) [emacs]

# [Q]
q           # Salir del modo copia (cancel) [vi]
PREFIX q    # Mostrar índices de paneles brevemente (display-panes)

# [R]
r           # Refrescar contenido desde el panel (refresh-from-pane)
R           # Alternar selección rectangular (rectangle-toggle)
PREFIX r    # Forzar redibujado del cliente (refresh-client)
Right       # Cursor derecha (cursor-right)
C-r         # Búsqueda incremental hacia atrás (search-backward) [emacs]

# [S]
Space       # Iniciar selección (begin-selection) [vi]
C-Space     # Iniciar selección (begin-selection) [emacs]
PREFIX Space    # Cambiar al siguiente layout predefinido (next-layout)
PREFIX s    # Elegir sesión interactivamente (choose-tree)
C-s         # Búsqueda incremental hacia adelante (search-forward) [emacs]

# [T]
T           # Saltar hacia atrás (antes del carácter) (jump-to-backward) [vi]
t           # Saltar hacia adelante (antes del carácter) (jump-to-forward) [vi]
PREFIX t    # Mostrar reloj en pantalla (clock-mode)

# [U]
Up          # Cursor arriba (cursor-up)
M-Up        # Media página arriba (halfpage-up) [emacs]
C-u         # Media página arriba (halfpage-up) [vi]
C-Up        # Scroll hacia arriba (scroll-up) [emacs]

# [V]
V           # Seleccionar línea completa (select-line) [vi]
v           # Alternar selección rectangular (rectangle-toggle) [vi]

# [W]
W           # Siguiente palabra separada por espacios (next-space) [vi]
w           # Siguiente palabra (next-word) [vi]
PREFIX w    # Elegir ventana interactivamente (choose-tree -Zw)
M-w         # Copiar selección y salir (copy-selection) [emacs]
C-w         # Copiar selección y salir (copy-selection) [emacs]

# [X]
X           # Establecer marca (set-mark)
PREFIX x    # Matar panel actual (kill-pane)
M-x         # Saltar a la marca (jump-to-mark) [emacs]

# [Y]
y           # Copiar selección y salir (copy-selection) [vi]
C-y         # Scroll hacia arriba (scroll-up) [emacs]

# [Z]
z           # Centrar cursor verticalmente (scroll-middle) [vi]
PREFIX z    # Maximizar/restaurar panel actual (resize-pane -Z)
PREFIX C-z  # Suspender cliente tmux (suspend-client)

# [<]
M-<         # Ir al inicio del historial (history-top) [emacs]

# [\]
PREFIX ]    # Pegar buffer (custom: si existe 'accum' lo pega, si no, pega normal)
PREFIX C-]  # Vaciar el buffer acumulador 'accum' (custom)
]           # (en modo copia no aplica)

# [^]
^           # Ir al primer carácter no blanco (back-to-indentation) [vi]

# [{]
{           # Párrafo anterior (previous-paragraph) [vi]
PREFIX {    # Intercambiar con panel anterior (swap-pane -U)
M-{         # Párrafo anterior (previous-paragraph) [emacs]

# [}]
}           # Siguiente párrafo (next-paragraph) [vi]
PREFIX }    # Intercambiar con panel siguiente (swap-pane -D)
M-}         # Siguiente párrafo (next-paragraph) [emacs]

# [~]
PREFIX ~    # Mostrar mensajes anteriores de tmux (show-messages)

# [Layouts]
PREFIX M-1  # Layout even-horizontal
PREFIX M-2  # Layout even-vertical
PREFIX M-3  # Layout main-horizontal
PREFIX M-4  # Layout main-horizontal-mirrored
PREFIX M-5  # Layout main-vertical
PREFIX M-6  # Layout main-vertical-mirrored
PREFIX M-7  # Layout tiled

# [Navegación configurada]
M-h / M-Left        # Panel izquierda
M-j / M-Down        # Panel abajo
M-k / M-Up          # Panel arriba
M-l / M-Right       # Panel derecha
M-H / M-S-Left      # Redimensionar izquierda 1
M-J / M-S-Down      # Redimensionar abajo 1
M-K / M-S-Up        # Redimensionar arriba 1
M-L / M-S-Right     # Redimensionar derecha 1
